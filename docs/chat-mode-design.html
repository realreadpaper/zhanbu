🔮 占卜多轮聊天模式

详细设计方案 v1.0 — 从一次性解读到 DeepSeek 式多轮对话

📋 1. 项目概述

将当前占卜平台的 AI 解读功能从一次性输出改造为多轮对话模式，让用户在获得首次解读后可以继续追问、深入探讨，实现类似 DeepSeek/ChatGPT 的聊天体验。

核心目标

  • 🎯 首次 AI 解读完成后，无缝进入聊天模式
  • 💬 用户可以针对占卜结果进行多轮追问
  • 🧠 AI 保持完整的对话上下文，理解用户意图
  • 📜 历史对话可保存、可查看、可继续
  • ⚡ 保持 SSE 流式响应，打字机效果

-- 聊天会话表 CREATE TABLE chat_sessions ( id BIGSERIAL PRIMARY KEY, user_id BIGINT NOT NULL REFERENCES users(id) ON DELETE CASCADE, record_id BIGINT NOT NULL REFERENCES divination_records(id) ON DELETE CASCADE, title VARCHAR(200) DEFAULT '', created_at TIMESTAMP DEFAULT NOW(), updated_at TIMESTAMP DEFAULT NOW() ); -- 聊天消息表 CREATE TABLE chat_messages ( id BIGSERIAL PRIMARY KEY, session_id BIGINT NOT NULL REFERENCES chat_sessions(id) ON DELETE CASCADE, role VARCHAR(20) NOT NULL, -- 'user' | 'assistant' content TEXT NOT NULL, created_at TIMESTAMP DEFAULT NOW() ); -- 索引 CREATE INDEX idx_chat_sessions_user ON chat_sessions(user_id); CREATE INDEX idx_chat_sessions_record ON chat_sessions(record_id); CREATE INDEX idx_chat_messages_session ON chat_messages(session_id, created_at);

后端 Chat Service 核心方法

// ChatService 核心接口 type ChatService struct { aiProvider AIProvider chatRepo ChatRepository divinationRepo DivinationRepository } // SendMessage 处理用户消息并返回 AI 回复流 func (s *ChatService) SendMessage( userID uint, sessionID uint, content string, ) (<-chan string, error) { // 1. 验证会话归属 session, _ := s.chatRepo.FindSessionByID(sessionID) // 2. 保存用户消息 s.chatRepo.CreateMessage(sessionID, "user", content) // 3. 加载对话历史 history, _ := s.chatRepo.GetRecentMessages(sessionID, MaxHistoryMessages) // 4. 加载占卜记录 record, _ := s.divinationRepo.FindByID(session.RecordID) // 5. 构建 messages 数组 messages := s.buildMessages(record, history) // 6. 调用 AI ch, _ := s.aiProvider.ChatCompletion(messages) // 7. 后台保存完整回复 go s.collectAndSave(ch, sessionID) return ch, nil }

前端 useChat Hook

// useChat.ts 核心逻辑 export function useChat(sessionId: number | null) { const [messages, setMessages] = useState<Message[]>([]) const [isStreaming, setIsStreaming] = useState(false) // 发送消息 const sendMessage = async (content: string) => { // 1. 添加用户消息到本地 setMessages(prev => [...prev, { role: 'user', content }]) // 2. 创建临时 AI 消息占位 setMessages(prev => [...prev, { role: 'assistant', content: '' }]) // 3. SSE 流式接收 setIsStreaming(true) chatStream(sessionId, content, (chunk) => { // 追加到 AI 消息 setMessages(prev => { const last = prev[prev.length - 1] return [...prev.slice(0, -1), { ...last, content: last.content + chunk }] }) }, () => setIsStreaming(false) ) } return { messages, sendMessage, isStreaming } }

⚠️ 12. 注意事项与风险

⚠️ 风险点

  • Token 成本：多轮对话会消耗更多 token，需监控用量
  • 上下文溢出：超长对话可能超出模型上下文限制
  • 存储增长：聊天消息表会持续增长
  • 并发压力：SSE 长连接占用服务器资源

✅ 应对措施

  • 滑动窗口：限制历史消息数量，控制 token
  • 消息清理：定期清理过期会话
  • 限流：每用户每分钟限制聊天次数
  • 监控：添加 AI 调用次数和 token 用量监控
